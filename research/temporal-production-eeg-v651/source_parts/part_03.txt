    trialTest = table(st, rp, true, ...
        'VariableNames', {'startSample','respSample','fixedWindowValid'});
    [dbA,okA] = compute_trialwise_causal_bandpower( ...
        causalA, trialTest, true(1,nCausal), nativeCausalFs, fs, cfg, 'fix500');
    [dbB,okB] = compute_trialwise_causal_bandpower( ...
        causalB, trialTest, true(1,nCausal), nativeCausalFs, fs, cfg, 'fix500');
    if ~okA || ~okB || any(abs(dbA-dbB) > 1e-10, 'all')
        error('v4:SelfTestCausalLeakage', ...
            'Post-window samples changed a causal fixed-window feature.');
    end
    causalC = causalA;
    causalC(:,rpNative:end) = causalC(:,rpNative:end) + 1e4;
    [dbFullA,okFullA] = compute_trialwise_causal_bandpower( ...
        causalA, trialTest, true(1,nCausal), nativeCausalFs, fs, cfg, 'full');
    [dbFullC,okFullC] = compute_trialwise_causal_bandpower( ...
        causalC, trialTest, true(1,nCausal), nativeCausalFs, fs, cfg, 'full');
    if ~okFullA || ~okFullC || any(abs(dbFullA-dbFullC) > 1e-10, 'all')
        error('v4:SelfTestCausalFullLeakage', ...
            'Response-or-later samples changed a causal full-window feature.');
    end
    fprintf('Synthetic self-tests passed.\n');
end

function inputT = write_run_manifest(cfg, status)
    % Validate input provenance before overwriting any run metadata.
    inputPath = fullfile(cfg.outRoot, 'input_manifest.csv');
    inputT = build_input_manifest(cfg);
    if exist(inputPath, 'file')
        previousInputT = readtable(inputPath, 'TextType','string');
        validate_input_manifest_unchanged(previousInputT, inputT);
    end

    manifest = struct();
    manifest.pipeline = 'run_ds004200_reproducible_v6_5_1_original_compatible';
    manifest.pipelineVersion = cfg.pipelineVersion;
    manifest.status = status;
    manifest.timestampUTC = char(datetime('now','TimeZone','UTC', ...
        'Format','yyyy-MM-dd''T''HH:mm:ss''Z'''));
    manifest.matlabVersion = version;
    manifest.computer = computer;
    manifest.rngAlgorithm = 'mt19937ar';
    manifest.masterSeed = cfg.masterSeed;
    manifest.config = cfg;
    manifest.dependencies = ver;
    codePath = [mfilename('fullpath') '.m'];
    manifest.codePath = codePath;
    manifest.codeSHA256 = hash_file_sha256(codePath);
    manifest.montagePath = cfg.montageFile;
    manifest.montageSHA256 = hash_file_sha256(cfg.montageFile);

    jsonPath = fullfile(cfg.outRoot, 'run_manifest.json');
    fid = fopen(jsonPath, 'w');
    if fid < 0, error('v6:ManifestWrite', 'Cannot write %s.', jsonPath); end
    cleanup = onCleanup(@() fclose(fid));
    fwrite(fid, jsonencode(manifest, 'PrettyPrint', true), 'char');
    clear cleanup;

    writetable(inputT, inputPath);

    statusPath = fullfile(cfg.outRoot, 'RUN_STATUS.txt');
    fid = fopen(statusPath, 'w');
    if fid < 0, error('v6:StatusWrite', 'Cannot write %s.', statusPath); end
    cleanup = onCleanup(@() fclose(fid));
    fprintf(fid, '%s\n', status);
    clear cleanup;
    if strcmp(status,'SUCCESS')
        successPath = fullfile(cfg.outRoot, 'SUCCESS');
        fid = fopen(successPath, 'w');
        if fid < 0, error('v6:SuccessWrite', 'Cannot create %s.', successPath); end
        cleanup = onCleanup(@() fclose(fid));
        fprintf(fid, 'Validated %s\n', manifest.timestampUTC);
        clear cleanup;
    end
end

function validate_input_manifest_unchanged(previousT, currentT)
    required = {'subject','inputType','path','bytes','sha256'};
    if ~all(ismember(required, previousT.Properties.VariableNames)) || ...
            ~all(ismember(required, currentT.Properties.VariableNames))
        error('v6:InputManifestSchema', 'Input manifest is missing required provenance columns.');
    end
    prev = previousT(:,required);
    cur = currentT(:,required);
    prev.subject = string(prev.subject); prev.inputType = string(prev.inputType);
    prev.path = string(prev.path); prev.sha256 = string(prev.sha256);
    cur.subject = string(cur.subject); cur.inputType = string(cur.inputType);
    cur.path = string(cur.path); cur.sha256 = string(cur.sha256);
    prev = sortrows(prev, {'subject','inputType','path'});
    cur = sortrows(cur, {'subject','inputType','path'});
    if height(prev) ~= height(cur) || ~isequaln(prev, cur)
        error('v6:InputProvenanceChanged', ...
            ['Current EEG/behavior inputs differ from the existing input_manifest.csv. ' ...
             'Use a new runLabel; checkpoints from mixed inputs are forbidden.']);
    end
end

function contract = build_checkpoint_contract(subID, cfg, inputManifest)
    codePath = [mfilename('fullpath') '.m'];
    contract.pipelineVersion = string(cfg.pipelineVersion);
    contract.codeSHA256 = string(hash_file_sha256(codePath));
    contract.montageSHA256 = string(hash_file_sha256(cfg.montageFile));
    contract.environment = software_environment_fingerprint(cfg);
    rows = inputManifest.subject == string(subID);
    S = inputManifest(rows, {'subject','inputType','bytes','sha256'});
    S.subject = string(S.subject);
    S.inputType = string(S.inputType);
    S.sha256 = string(S.sha256);
    S = sortrows(S, {'inputType'});
    if isempty(S)
        error('v6:CheckpointInputContract', '%s has no input-manifest rows.', subID);
    end
    contract.inputType = string(S.inputType(:));
    contract.inputBytes = double(S.bytes(:));
    contract.inputSHA256 = string(S.sha256(:));
end

function fp = software_environment_fingerprint(cfg)
    fp.matlabVersion = string(version);
    fp.computer = string(computer);
    names = {'eeglab','pop_eegfiltnew','pop_runica','iclabel'};
    for i = 1:numel(names)
        nm = names{i};
        p = which(nm);
        fieldPath = matlab.lang.makeValidName([nm '_path']);
        fieldHash = matlab.lang.makeValidName([nm '_sha256']);
        fp.(fieldPath) = string(p);
        if isempty(p)
            if strcmp(nm,'iclabel') && ~cfg.doICA
                fp.(fieldHash) = "not_required";
            else
                fp.(fieldHash) = "missing";
            end
        elseif exist(p,'file') == 2
            fp.(fieldHash) = string(hash_file_sha256(p));
        else
            fp.(fieldHash) = "non_file_resolution";
        end
    end
end

function T = build_input_manifest(cfg)
    T = table();
    row = 0;
    for s = 1:numel(cfg.subjects)
        sub = cfg.subjects{s};
        paths = { ...
            fullfile(cfg.root, sub, 'eeg', [sub '_task-temporalscaling_eeg.set']), ...
            fullfile(cfg.root, sub, 'eeg', [sub '_task-temporalscaling_eeg.fdt']), ...
            fullfile(cfg.root, sub, 'beh', [sub '_task-temporalscaling_beh.tsv'])};
        types = {'EEG_SET','EEG_FDT','BEHAVIOR_TSV'};
        for i = 1:numel(paths)
            if ~exist(paths{i}, 'file')
                if strcmp(types{i},'EEG_FDT'), continue; end
                error('v4:MissingInputManifestFile', 'Missing required input: %s', paths{i});
            end
            info = dir(paths{i});
            row = row + 1;
            T.subject(row,1) = string(sub);
            T.inputType(row,1) = string(types{i});
            T.path(row,1) = string(paths{i});
            T.bytes(row,1) = info.bytes;
            T.modified(row,1) = string(info.date);
            T.sha256(row,1) = string(hash_file_sha256(paths{i}));
        end
    end
end

function digest = hash_file_sha256(path)
    quoted = strrep(path, '"', '""');
    if ispc
        [status, output] = system(sprintf('certutil -hashfile "%s" SHA256', quoted));
    else
        [status, output] = system(sprintf('sha256sum "%s"', quoted));
    end
    if status ~= 0
        error('v4:HashFailure', 'SHA-256 failed for %s: %s', path, output);
    end
    token = regexp(output, '(?i)[0-9a-f]{64}', 'match', 'once');
    if isempty(token)
        error('v4:HashParseFailure', 'Could not parse SHA-256 for %s.', path);
    end
    digest = lower(token);
end

function [EEG, causalData, causalGoodMask, causalFs, prepQC, rawGoodMask] = ...
        preprocess_subject_eeg(EEG, subID, cfg)
    prepQC = struct('badChannelCount',0,'badChannelLabels',"", ...
        'rejectedICCount',0,'rejectedICLabels',"",'afzMode',"", ...
        'mastoidStdUv',[nan nan],'mastoidStepFraction',[nan nan]);
    EEG = eeg_checkset(EEG);
    if EEG.trials ~= 1
        error('v4:EpochedInput', '%s input must be continuous.', subID);
    end

    rawLabels = get_chan_labels(EEG);
    if any(strlength(strtrim(rawLabels)) == 0) || ...
            numel(unique(lower(strtrim(rawLabels)))) ~= numel(rawLabels)
        error('v4:InvalidRawLabels', '%s has blank or duplicate channel labels.', subID);
    end
    recordedScalp = setdiff(string(cfg.scalpLabels), "AFz", 'stable');
    required = lower([recordedScalp, string(cfg.auxLabels)]);
    if ~all(ismember(required, lower(rawLabels)))
        missing = required(~ismember(required, lower(rawLabels)));
        error('v4:MissingRawChannels', '%s missing required channels: %s', subID, strjoin(missing, ', '));
    end
    EEG = apply_official_layout(EEG, cfg, subID);

    % Validate LM/RM on a 200-Hz, 0.5-45-Hz QC copy. The public BIDS
    % recordings can contain very large slow/DC mastoid drift at the native
    % sampling rate; applying absolute-amplitude QC before high-pass
    % filtering can therefore reject physiologically usable mastoids. The
    % raw copy is retained only for the independent 50-Hz line-noise check.
    EEGmastoidQC = EEG;
    if abs(EEGmastoidQC.srate - cfg.targetFs) > 1e-6
        EEGmastoidQC = pop_resample(EEGmastoidQC, cfg.targetFs);
    end
    EEGmastoidQC = pop_eegfiltnew(EEGmastoidQC, ...
        cfg.broadBandHz(1), cfg.broadBandHz(2));
    EEGmastoidQC = eeg_checkset(EEGmastoidQC);
    [mastoidStd, mastoidStep] = validate_mastoid_channels( ...
        EEGmastoidQC, EEG, subID, cfg);
    prepQC.mastoidStdUv = mastoidStd;
    prepQC.mastoidStepFraction = mastoidStep;
    clear EEGmastoidQC;

    % The acquisition used AFz as its online reference. Official BIDS files
    % append a zero-valued AFz placeholder. Also support source-like files
    % in which AFz is absent by explicitly reconstructing it with the
    % official neuroscan37 location during linked-mastoid rereferencing.
    rawLabels = get_chan_labels(EEG);
    afzIdx = find(strcmpi(rawLabels,"AFz"));
    if isempty(afzIdx)
        afzLoc = official_afz_chanloc(cfg);
        EEG.data(end+1,:) = zeros(1,EEG.pnts,class(EEG.data));
        EEG.nbchan = EEG.nbchan + 1;
        EEG.chanlocs(end+1) = afzLoc;
        prepQC.afzMode = "reconstructed_from_acquisition_reference";
    elseif isscalar(afzIdx)
        validate_afz_placeholder(EEG, afzIdx, subID, cfg);
        prepQC.afzMode = "validated_bids_placeholder";
    else
        error('v4:DuplicateAFz', '%s has more than one AFz channel.', subID);
    end
    EEG = linked_mastoid_rereference(EEG, cfg, subID);

    labelsNow = get_chan_labels(EEG);
    [present, scalpIdx] = ismember(lower(string(cfg.scalpLabels)), lower(labelsNow));
    if ~all(present)
        error('v4:IncompleteScalpMontage', '%s lost scalp channels during rereferencing.', subID);
    end
    EEG = pop_select(EEG, 'channel', scalpIdx);
    EEG = eeg_checkset(EEG);
    causalFs = EEG.srate;
    scalpTemplate = EEG.chanlocs;
    validate_scalp_layout(EEG.chanlocs, string(cfg.scalpLabels), subID);

    % Build the native-rate artifact mask from a strictly causal 0.5-45-Hz
    % QC representation. This removes the large slow/DC mastoid-reference
    % drift without using future samples, while the unfiltered native data
    % remain the input to the causal prediction branch.
    [causalQcData, causalQcBurnIn, nativeFiniteGood] = causal_qc_bandpass( ...
        double(EEG.data), cfg, causalFs, subID);
    preliminaryBadIdx = detect_bad_scalp_channels(causalQcData, cfg);
    if numel(preliminaryBadIdx) > cfg.maxBadScalpChannels
        error('v4:TooManyRawBadChannels', '%s has %d preliminary bad scalp channels.', ...
            subID, numel(preliminaryBadIdx));
    end
    maskChannels = setdiff(1:EEG.nbchan, preliminaryBadIdx);
    causalGoodMask = make_causal_artifact_mask( ...
        causalQcData(maskChannels,:), cfg, causalFs);
    % Preserve raw nonfinite samples as permanently invalid even though the
    % QC representation is causally repaired before filtering.
    causalGoodMask = causalGoodMask & nativeFiniteGood(:);
    causalGoodMask(1:min(causalQcBurnIn,numel(causalGoodMask))) = false;
    clear causalQcData;
    preliminaryBadLabels = string({EEG.chanlocs(preliminaryBadIdx).labels});
    if ~isempty(preliminaryBadIdx)
        EEG = pop_select(EEG, 'nochannel', preliminaryBadIdx);
    end
    EEGcausal = EEG;
    EEGcausal.data = repair_masked_samples_causal( ...
        double(EEGcausal.data), causalGoodMask, subID);
    EEG.data = interpolate_masked_samples(double(EEG.data), causalGoodMask, subID);

    % Only the descriptive branch is resampled. The causal prediction branch
    % remains at the acquisition rate so an offline anti-alias filter cannot
    % mix post-response samples into its predictors.
    if abs(EEG.srate - cfg.targetFs) > 1e-6
        nativeFs = EEG.srate;
        EEG = pop_resample(EEG, cfg.targetFs);
        rawGoodMask = resample_good_mask_conservative( ...
            causalGoodMask, nativeFs, EEG.srate, EEG.pnts);
    else
        rawGoodMask = causalGoodMask;
    end

    if cfg.doNotchFilter
        EEG = pop_eegfiltnew(EEG, cfg.notchHz-1, cfg.notchHz+1, [], 1);
    end
    EEG = pop_eegfiltnew(EEG, cfg.broadBandHz(1), cfg.broadBandHz(2));
