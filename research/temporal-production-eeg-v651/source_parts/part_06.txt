    end

    % PRIMARY adjacency: reproduce original analysis code by selecting each
    % production block TYPE (1/2/3), preserving row order, and pairing adjacent
    % rows in that condition sequence. This can cross a physical 10-trial block
    % boundary when the same block type reappears later; that is intentional and
    % exactly why strict protocol-block fields are exported separately.
    nRows = height(behMaster);
    behMaster.prevProducedDuration = nan(nRows,1);
    behMaster.prevRelError = nan(nRows,1);
    behMaster.prevFeedbackSigned = nan(nRows,1);
    behMaster.prevBehaviorValid = false(nRows,1);
    behMaster.nextProducedDuration = nan(nRows,1);
    behMaster.nextRelError = nan(nRows,1);
    behMaster.nextRelDelta = nan(nRows,1);
    behMaster.correctionGain = nan(nRows,1);
    behMaster.nextBehaviorValid = false(nRows,1);
    behMaster.paperConditionMeanProducedDuration = nan(nRows,1);
    behMaster.paperNextAdjustmentSec = nan(nRows,1);

    % STRICT sensitivity adjacency: never crosses a reconstructed physical block.
    behMaster.strictPrevProducedDuration = nan(nRows,1);
    behMaster.strictPrevRelError = nan(nRows,1);
    behMaster.strictPrevFeedbackSigned = nan(nRows,1);
    behMaster.strictPrevBehaviorValid = false(nRows,1);
    behMaster.strictNextProducedDuration = nan(nRows,1);
    behMaster.strictNextRelError = nan(nRows,1);
    behMaster.strictNextRelDelta = nan(nRows,1);
    behMaster.strictCorrectionGain = nan(nRows,1);
    behMaster.strictNextBehaviorValid = false(nRows,1);
    behMaster.primaryAdjacencyMode = repmat("original_condition_sequence", nRows, 1);

    subjects = unique(behMaster.subject, 'stable');
    for s = 1:numel(subjects)
        prodMask = behMaster.subject == subjects(s) & behMaster.task == "production";

        % Original-paper condition sequence (block types 1/2/3).
        for code = 1:3
            idx = find(prodMask & behMaster.block == code);
            [~,ord] = sort(behMaster.conditionTrialSeq(idx));
            idx = idx(ord);
            if isempty(idx), continue; end
            condMean = mean(behMaster.producedDuration(idx));
            behMaster.paperConditionMeanProducedDuration(idx) = condMean;
            for k = 2:numel(idx)
                cur = idx(k); prev = idx(k-1);
                if behMaster.behaviorDurationValid(prev)
                    behMaster.prevProducedDuration(cur) = behMaster.producedDuration(prev);
                    behMaster.prevRelError(cur) = behMaster.relError(prev);
                    behMaster.prevFeedbackSigned(cur) = behMaster.feedbackSigned(prev);
                    behMaster.prevBehaviorValid(cur) = true;
                end
            end
            for k = 1:(numel(idx)-1)
                cur = idx(k); nxt = idx(k+1);
                % Exact paper-style adjustment variable, exported for audit only.
                if isfinite(behMaster.producedDuration(nxt)) && isfinite(condMean)
                    behMaster.paperNextAdjustmentSec(cur) = behMaster.producedDuration(nxt) - condMean;
                end
                if behMaster.behaviorDurationValid(nxt)
                    behMaster.nextProducedDuration(cur) = behMaster.producedDuration(nxt);
                    behMaster.nextRelError(cur) = behMaster.relError(nxt);
                    behMaster.nextBehaviorValid(cur) = true;
                    if behMaster.behaviorDurationValid(cur)
                        behMaster.nextRelDelta(cur) = behMaster.nextRelError(cur) - behMaster.relError(cur);
                        behMaster.correctionGain(cur) = abs(behMaster.relError(cur)) - abs(behMaster.nextRelError(cur));
                    end
                end
            end
        end

        % Strict true-block sensitivity.
        blocks = unique(behMaster.protocolBlock(prodMask & isfinite(behMaster.protocolBlock)));
        for bi = 1:numel(blocks)
            b = blocks(bi);
            idx = find(prodMask & behMaster.protocolBlock == b);
            [~,ord] = sort(behMaster.withinBlockSeq(idx));
            idx = idx(ord);
            for k = 2:numel(idx)
                cur = idx(k); prev = idx(k-1);
                if behMaster.behaviorDurationValid(prev)
                    behMaster.strictPrevProducedDuration(cur) = behMaster.producedDuration(prev);
                    behMaster.strictPrevRelError(cur) = behMaster.relError(prev);
                    behMaster.strictPrevFeedbackSigned(cur) = behMaster.feedbackSigned(prev);
                    behMaster.strictPrevBehaviorValid(cur) = true;
                end
            end
            for k = 1:(numel(idx)-1)
                cur = idx(k); nxt = idx(k+1);
                if behMaster.behaviorDurationValid(nxt)
                    behMaster.strictNextProducedDuration(cur) = behMaster.producedDuration(nxt);
                    behMaster.strictNextRelError(cur) = behMaster.relError(nxt);
                    behMaster.strictNextBehaviorValid(cur) = true;
                    if behMaster.behaviorDurationValid(cur)
                        behMaster.strictNextRelDelta(cur) = ...
                            behMaster.strictNextRelError(cur) - behMaster.relError(cur);
                        behMaster.strictCorrectionGain(cur) = ...
                            abs(behMaster.relError(cur)) - abs(behMaster.strictNextRelError(cur));
                    end
                end
            end
        end
    end

    validate_behavior_adjacency_contracts(behMaster, cfg);
end

function ids = reconstruct_task_protocol_blocks(trialNum, blockType, subID, taskName, cfg)
    trialNum = double(trialNum(:));
    blockType = double(blockType(:));
    ids = nan(size(trialNum));
    if isempty(trialNum), return; end
    blockCounter = 0;
    prevTrial = NaN;
    prevType = NaN;
    for i = 1:numel(trialNum)
        if i == 1 || trialNum(i) == 1
            blockCounter = blockCounter + 1;
        else
            if trialNum(i) ~= prevTrial + 1 || blockType(i) ~= prevType
                error('v64:ProtocolBlockSequence', ...
                    ['%s %s trial sequence cannot be reconstructed safely at task-row %d ' ...
                     '(previous trial/type %.0f/%.0f, current %.0f/%.0f).'], ...
                    subID, taskName, i, prevTrial, prevType, trialNum(i), blockType(i));
            end
        end
        ids(i) = blockCounter;
        prevTrial = trialNum(i);
        prevType = blockType(i);
    end
    if any(~isfinite(ids)) || max(ids) * cfg.trialsPerProtocolBlock ~= numel(trialNum)
        error('v64:ProtocolBlockCount', ...
            '%s %s rows do not form complete %d-trial blocks.', ...
            subID, taskName, cfg.trialsPerProtocolBlock);
    end
end

function validate_behavior_adjacency_contracts(T, cfg)
    subs = unique(T.subject, 'stable');
    tol = 1e-12;
    for si = 1:numel(subs)
        prod = T.subject == subs(si) & T.task == "production";

        % PRIMARY: exact adjacency in the original-paper condition sequence.
        for code = 1:3
            idx = find(prod & T.block == code);
            [~,ord] = sort(T.conditionTrialSeq(idx)); idx = idx(ord);
            if isempty(idx), continue; end
            if T.prevBehaviorValid(idx(1)) || isfinite(T.prevRelError(idx(1)))
                error('v64:PrimaryAdjacencyStart', ...
                    '%s condition %d first row inherited a previous primary outcome.', subs(si), code);
            end
            if T.nextBehaviorValid(idx(end)) || isfinite(T.nextRelError(idx(end)))
                error('v64:PrimaryAdjacencyEnd', ...
                    '%s condition %d last row inherited a next primary outcome.', subs(si), code);
            end
            for k = 2:numel(idx)
                cur = idx(k); prev = idx(k-1);
                if T.behaviorDurationValid(prev)
                    if ~T.prevBehaviorValid(cur) || ...
                            abs(T.prevProducedDuration(cur)-T.producedDuration(prev)) > tol || ...
                            abs(T.prevRelError(cur)-T.relError(prev)) > tol
                        error('v64:PrimaryPreviousPairContract', ...
                            '%s condition %d primary previous-pair mismatch.', subs(si), code);
                    end
                elseif T.prevBehaviorValid(cur)
                    error('v64:PrimaryPreviousValidityContract', ...
                        '%s condition %d inherited an invalid previous duration.', subs(si), code);
                end
            end
            for k = 1:(numel(idx)-1)
                cur = idx(k); nxt = idx(k+1);
                if T.behaviorDurationValid(nxt)
                    if ~T.nextBehaviorValid(cur) || ...
                            abs(T.nextProducedDuration(cur)-T.producedDuration(nxt)) > tol || ...
                            abs(T.nextRelError(cur)-T.relError(nxt)) > tol
                        error('v64:PrimaryNextPairContract', ...
                            '%s condition %d primary next-pair mismatch.', subs(si), code);
                    end
                elseif T.nextBehaviorValid(cur)
                    error('v64:PrimaryNextValidityContract', ...
                        '%s condition %d inherited an invalid next duration.', subs(si), code);
                end
            end
        end

        % STRICT: exact adjacency inside each reconstructed 10-trial block.
        blocks = unique(T.protocolBlock(prod & isfinite(T.protocolBlock)));
        for bi = 1:numel(blocks)
            idx = find(prod & T.protocolBlock == blocks(bi));
            [~,ord] = sort(T.withinBlockSeq(idx)); idx = idx(ord);
            if numel(idx) ~= cfg.trialsPerProtocolBlock
                error('v64:StrictBlockSizeContract', ...
                    '%s strict block %d is incomplete.', subs(si), blocks(bi));
            end
            if T.strictPrevBehaviorValid(idx(1)) || isfinite(T.strictPrevRelError(idx(1))) || ...
                    T.strictNextBehaviorValid(idx(end)) || isfinite(T.strictNextRelError(idx(end)))
                error('v64:StrictBoundaryContract', ...
                    '%s strict block %d crossed a boundary.', subs(si), blocks(bi));
            end
            for k = 2:numel(idx)
                cur = idx(k); prev = idx(k-1);
                if T.behaviorDurationValid(prev)
                    if ~T.strictPrevBehaviorValid(cur) || ...
                            abs(T.strictPrevRelError(cur)-T.relError(prev)) > tol
                        error('v64:StrictPreviousPairContract', ...
                            '%s strict block %d previous-pair mismatch.', subs(si), blocks(bi));
                    end
                elseif T.strictPrevBehaviorValid(cur)
                    error('v64:StrictPreviousValidityContract', ...
                        '%s strict block %d inherited invalid previous behavior.', subs(si), blocks(bi));
                end
            end
            for k = 1:(numel(idx)-1)
                cur = idx(k); nxt = idx(k+1);
                if T.behaviorDurationValid(nxt)
                    if ~T.strictNextBehaviorValid(cur) || ...
                            abs(T.strictNextRelError(cur)-T.relError(nxt)) > tol
                        error('v64:StrictNextPairContract', ...
                            '%s strict block %d next-pair mismatch.', subs(si), blocks(bi));
                    end
                elseif T.strictNextBehaviorValid(cur)
                    error('v64:StrictNextValidityContract', ...
                        '%s strict block %d inherited invalid next behavior.', subs(si), blocks(bi));
                end
            end
        end
    end
end

function Q = audit_behavior_master(behMaster, cfg)
    subjects = unique(behMaster.subject, 'stable');
    Q = table();
    for s = 1:numel(subjects)
        sub = subjects(s);
        P = behMaster(behMaster.subject == sub & behMaster.task == "production", :);
        R = behMaster(behMaster.subject == sub & behMaster.task == "perception", :);
        row = height(Q) + 1;
        Q.subject(row,1) = sub;
        Q.nProduction(row,1) = height(P);
        Q.nPerception(row,1) = height(R);
        Q.nValidProductionDuration(row,1) = sum(P.behaviorDurationValid);
        Q.nInvalidProductionDuration(row,1) = sum(~P.behaviorDurationValid);
        Q.nProductionProtocolBlocks(row,1) = numel(unique(P.protocolBlock(isfinite(P.protocolBlock))));
        Q.nPerceptionProtocolBlocks(row,1) = numel(unique(R.protocolBlock(isfinite(R.protocolBlock))));
        Q.trialsPerProtocolBlock(row,1) = cfg.trialsPerProtocolBlock;
        Q.productionBlockIntegrity(row,1) = protocol_block_integrity(P, cfg);
        Q.perceptionBlockIntegrity(row,1) = protocol_block_integrity(R, cfg);
        Q.primaryAdjacencyMode(row,1) = "original_condition_sequence";
        Q.nPrimaryPrevCrossProtocol(row,1) = sum(P.prevBehaviorValid & ...
            isfinite(P.protocolBlock) & isfinite(P.withinBlockSeq) & P.withinBlockSeq == 1);
        Q.nPrimaryNextCrossProtocol(row,1) = sum(P.nextBehaviorValid & ...
            isfinite(P.protocolBlock) & isfinite(P.withinBlockSeq) & ...
            P.withinBlockSeq == cfg.trialsPerProtocolBlock);
        for code = 1:3
            Q.("nProductionBlockType" + string(code))(row,1) = sum(P.block == code);
        end
        if ~isempty(P)
            schemes = unique(P.feedbackCodeScheme);
            Q.feedbackCodeScheme(row,1) = strjoin(schemes, ';');
        else
            Q.feedbackCodeScheme(row,1) = "missing";
        end
        Q.expectedProductionTrials(row,1) = cfg.expectedProductionTrials;
        Q.expectedPerceptionTrials(row,1) = cfg.expectedPerceptionTrials;
        Q.productionCountDeviation(row,1) = height(P) - cfg.expectedProductionTrials;
        Q.perceptionCountDeviation(row,1) = height(R) - cfg.expectedPerceptionTrials;
        if height(P) ~= cfg.expectedProductionTrials || height(R) ~= cfg.expectedPerceptionTrials
            warning('v64:ProtocolTrialCountDeviation', ...
                '%s has %d production and %d perception rows (protocol expectation %d/%d).', ...
                sub, height(P), height(R), cfg.expectedProductionTrials, cfg.expectedPerceptionTrials);
        end
    end
end

function validate_protocol_blocks(T, subID, cfg)
    tasks = ["production","perception"];
    for ti = 1:numel(tasks)
        M = T(T.task == tasks(ti), :);
        if isempty(M), continue; end
        blocks = unique(M.protocolBlock(isfinite(M.protocolBlock)));
        for bi = 1:numel(blocks)
            B = M(M.protocolBlock == blocks(bi), :);
            if height(B) ~= cfg.trialsPerProtocolBlock || ...
                    numel(unique(B.withinBlockSeq)) ~= cfg.trialsPerProtocolBlock || ...
                    ~isequal(sort(B.withinBlockSeq), (1:cfg.trialsPerProtocolBlock)')
                error('v64:ProtocolBlockIntegrity', ...
                    '%s %s protocol block %d is not an intact %d-trial block.', ...
                    subID, tasks(ti), blocks(bi), cfg.trialsPerProtocolBlock);
            end
            if numel(unique(B.tempo)) ~= 1 || numel(unique(B.conditionCode)) ~= 1
                error('v64:ProtocolBlockConditionContract', ...
                    '%s %s protocol block %d spans multiple tempo/condition codes.', ...
                    subID, tasks(ti), blocks(bi));
            end
        end
    end
end
