function results = run_ds004200_reproducible_v6_5_1_original_compatible(userCfg)
%% RUN_DS004200_REPRODUCIBLE_V6_5_1_ORIGINAL_COMPATIBLE
% =========================================================================
% MANUSCRIPT-SAFE, ENDPOINT-GATED SCALP-ONLY EEG PIPELINE FOR OPENNEURO ds004200
% =========================================================================
%
% This callable pipeline replaces the artifact-contaminated v3 and prior v5
% script. It uses a canonical 32-channel scalp montage, linked-mastoid
% reference, explicit ICA/ICLabel component subtraction, applied trial QC,
% label-keyed topographies, grouped nested LOSO prediction, deterministic
% random streams, subject-balanced metrics, and publication-grade exports.
%
% IMPORTANT SCIENTIFIC CONSTRAINTS
%   * Auxiliary channels never enter scalp features, PSDs, interpolation,
%     topographies, or a scalp-average reference.
%   * No silent signal-processing fallback is permitted.
%   * Current-error models never include current produced duration as a predictor,
%     because relative error is algebraically defined from that value. Pre-start
%     EEG features use only samples preceding trial onset; model evaluation is
%     restricted to prespecified valid-duration trials.
%   * Primary previous/next outcomes reproduce the original authors' condition-sequence
%     analysis within subject x official production block type (1/2/3). A separate
%     strict sensitivity analysis uses reconstructed 10-trial protocol blocks.
%   * Bad-channel detection and ICA are unsupervised subject-level preprocessing
%     estimated from the recording; trial-level causal features themselves never
%     use post-window samples. This is leakage-controlled, not an online-adaptive pipeline.
%   * Corrected results may be weaker than v3. Validity is the objective;
%     this code does not optimize for a favorable manuscript result.
%
% Required:
%   - MATLAB R2022b or later
%   - EEGLAB 2024.x or later
%   - Signal Processing Toolbox and Statistics and Machine Learning Toolbox
%   - ICLabel plugin (required when cfg.doICA=true; default)
%
% =========================================================================

if nargin < 1 || isempty(userCfg)
    userCfg = struct();
end
if ~isstruct(userCfg)
    error('v4:InvalidConfig', 'userCfg must be a scalar struct.');
end

%% ===================== CONFIG =====================
cfg = struct();

cfg.pipelineVersion = '6.5.1';
cfg.masterSeed = 20260810;
cfg.runLabel = 'v6_5_1_original_compatible';
% Continue across subject-specific failures so the ledger captures every subject.
cfg.failOnSubjectError = false;
% Minimum analyzable cohort, not a target N. Actual included/endpoint N is always exported.
cfg.minimumIncludedSubjects = 12;
cfg.minimumPredictionSubjects = 12;
cfg.minimumSensitivitySubjects = 8;
cfg.minimumTopoSubjects = 12;
cfg.requireEmptyOutput = true;
cfg.resumeIncompleteRun = true;
cfg.checkpointSubjects = true;

% ---- CHANGE THESE IF NEEDED ----
cfg.root = 'F:\time perception 3\data';
cfg.eeglabPath = 'C:\eeglab2024.0';
cfg.montageFile = fullfile(fileparts(mfilename('fullpath')), 'neuroscan37_official.locs');

% Output folder. Use a new runLabel for every scientifically distinct run.
cfg.outRoot = '';

% Subjects. Empty = auto-detect all sub-*.
cfg.subjects = {};

% EEG processing
cfg.targetFs = 200;
cfg.broadBandHz = [0.5 45];
cfg.bandFilterOrder = 4;
cfg.causalImpulseTailEnergy = 1e-6;
cfg.causalImpulseMaxSec = 60;
cfg.doNotchFilter = false; % 45-Hz low-pass already excludes 50-Hz line noise
cfg.notchHz = 50;
cfg.referenceMode = 'linked_mastoids';
cfg.mastoidLabels = {'LM','RM'};
cfg.auxLabels = {'LM','RM','LEOG','REOG','VEOG'};
cfg.afzPlaceholderMaxStdUv = 1e-6;
cfg.afzRadius = 0.38333; % official neuroscan37.locs
cfg.afzCoordinateTolerance = 0.01;
cfg.afzThetaToleranceDeg = 0.5;
cfg.mastoidStdRatioRange = [0.10 10.0];
cfg.mastoidMaxStepFraction = 0.01;
cfg.mastoidMaxExtremeFraction = 0.01;
cfg.mastoidMaxLineNoiseRatio = 20;
cfg.scalpLabels = {'Fp1','Fp2','F7','F3','Fz','F4','F8', ...
    'FC5','FC1','FCz','FC2','FC6','T7','C3','Cz','C4','T8', ...
    'CP5','CP1','CPz','CP2','CP6','P7','P3','Pz','P4','P8', ...
    'POz','O1','Oz','O2','AFz'};

% Deterministic bad-channel cleaning and ICA
cfg.doICA = true;
cfg.badChannelRobustZ = 7;
cfg.badChannelMinStdUv = 0.10;
cfg.maxBadScalpChannels = 4;
cfg.icaHighpassHz = 1.0;
cfg.icaMuscleThreshold = 0.80;
cfg.icaEyeThreshold = 0.80;
cfg.icaLineThreshold = 0.90;
cfg.icaChannelNoiseThreshold = 0.90;
cfg.maxRejectedICFraction = 0.60;

% ROI for prediction
cfg.roiLabels = {'FCz','Cz','CPz','FC1','FC2','CP1','CP2'};

% Artifact QC
cfg.artifactAbsUv = 150;
cfg.artifactStepUv = 50;
cfg.artifactExpandSec = 0.25;
cfg.minTrialGoodFraction = 0.90;
cfg.minBaselineGoodFraction = 0.90;
cfg.maxMissingFractionPerChannel = 0.01;

% Event grammar
cfg.prod.startCodes = [12 22 32];
cfg.prod.respCodes  = [13 23 33];
cfg.prod.fbCorrect  = [14 24 34];
cfg.prod.fbEarly    = [15 25 35];
cfg.prod.fbLate     = [16 26 36];

% Known target intervals
cfg.targetDurationsSec = [0.800 1.650 2.500];

% Trial and alignment settings
cfg.baselineSec = 0.300;
% Original-paper production-duration QC (analysis_02_beh.m):
% lower bound 0.2 s for every condition; upper bound = 5 x target duration.
cfg.minTrialDurSec = 0.200;
cfg.maxTrialDurMultiplier = 5.0;
cfg.maxTrialDurSecByTempo = cfg.targetDurationsSec .* cfg.maxTrialDurMultiplier; % [4.0 8.25 12.5] s
cfg.maxDurationMismatchSec = 0.050;
cfg.maxRealignmentMismatchSec = 0.050; % match tolerance used directly inside alignment
cfg.skipPenaltySec = 0.075;
% Alignment is duration/order/tempo only. Feedback is reserved for independent QC.
cfg.minAlignmentCoverage = 0.80;      % hard subject-level gate
cfg.minAlignmentCoverageWarn = 0.90;  % warning band above the hard gate
cfg.minFeedbackValidationTrials = 20;
cfg.maxFeedbackDisagreementRate = 0.05;
% Dataset-protocol audit. Official BIDS metadata defines `block` as BLOCK TYPE:
% 1/2/3 = production fast/medium/slow; 4/5/6 = perception fast/medium/slow.
% The original behavioural analysis concatenated trials within each block type
% (condition) when forming trial-n -> trial-n+1 adjustment pairs. That exact
% condition-sequence adjacency is the PRIMARY analysis here. True 10-trial
% protocol blocks are reconstructed separately from the official `trial` column
% and retained for a stricter sensitivity analysis.
cfg.expectedProductionTrials = 180;
cfg.expectedPerceptionTrials = 180;
cfg.trialsPerProtocolBlock = 10;
cfg.primaryAdjacencyMode = 'original_condition_sequence';
cfg.runStrictBlockSensitivity = true;
cfg.requireOfficialProductionFeedback = true;

% Prediction windows
cfg.fixedEarlyWinSec = [0.000 0.500]; % descriptive/supplemental only
cfg.prestartPredictionSec = 0.500; % EEG feature window is outcome-independent; analysis cohort uses prespecified duration validity
cfg.minPrestartGoodFraction = 0.95;
% Causal filters may use a small amount of strictly past-only repaired history;
% requiring 100% clean multi-second burn-in was unnecessarily exclusionary.
cfg.minCausalHistoryGoodFraction = 0.95;
cfg.earlyPct = [0 40];
cfg.midPct   = [20 60];
cfg.fullPct  = [0 100];

% Topo/PSD windows
cfg.windows = struct();
cfg.windows.fix500.type = 'fixed';
cfg.windows.fix500.sec = [0.000 0.500];
cfg.windows.early.type = 'pct';
cfg.windows.early.pct = [0 40];
cfg.windows.mid.type = 'pct';
cfg.windows.mid.pct = [20 60];
cfg.windows.full.type = 'pct';
cfg.windows.full.pct = [0 100];
cfg.windowNames = {'fix500','early','mid','full'};

% Shared band definitions. Delta is descriptive only; primary causal models use theta-gamma.
cfg.predBands = [
    1 4;
    4 8;
    8 12;
    12 15;
    15 20;
    20 30;
    30 45
];
cfg.predBandNames = {'delta','theta','alpha','smr','beta_low','beta_mid','gamma'};

% Neurofigure bands: required paper figure bands
cfg.topoBandNames = {'theta','alpha','smr','beta_low','beta_mid','gamma'};

% Split for high/low behavioural topomap contrasts
cfg.splitMode = 'tertile';   % 'tertile' or 'median'

% PSD settings
cfg.psdFreqRange = [4 45]; % 0.5-s PSD is not used to claim delta-band resolution
cfg.psdNfft = 512;
cfg.psdFixedWindowSec = 0.500; % equal-length, trial-wise PSD only

% Prediction settings
cfg.minTrialsPerSubject = 30;
cfg.lambdaGrid = [0.001 0.01 0.1 1 10 100 1000];
cfg.innerFolds = 5;
cfg.nPermutation = 9999;
cfg.nBootstrap = 5000;
cfg.maxExactSignFlipSubjects = 20;

% Negative controls
cfg.doNegativeControls = true;
cfg.nShuffle = 1000;

% Main figure models
cfg.currentBaselineModel = 'B1_history';
cfg.currentMainEEGModel = 'C3_prestart_bandPower6';
cfg.nextBaselineModel = 'B1_currentBehavior';
cfg.nextMainEEGModel = 'N4_causalFull_bandPower6';

% Topographic inference and graphics
cfg.topoClusterFormingAlpha = 0.05;
cfg.topoPermutations = 9999;
cfg.topoNearestNeighbours = 4;
cfg.minTopoTrialsPerCondition = 5;
cfg.figureResolution = 600;
cfg.runSelfTests = true;

unknownCfg = setdiff(fieldnames(userCfg), fieldnames(cfg));
if ~isempty(unknownCfg)
    error('v6:UnknownConfigField', 'Unknown userCfg field(s): %s', strjoin(unknownCfg, ', '));
end
cfg = merge_structs(cfg, userCfg);
if isempty(cfg.outRoot)
    cfg.outRoot = fullfile(cfg.root, 'derivatives', cfg.runLabel);
end

validate_config_v6(cfg);
rng(cfg.masterSeed, 'twister');

ensure_dir(cfg.outRoot);
ensure_dir(fullfile(cfg.outRoot, 'tables'));
ensure_dir(fullfile(cfg.outRoot, 'figures'));
ensure_dir(fullfile(cfg.outRoot, 'mat'));
ensure_dir(fullfile(cfg.outRoot, 'logs'));
ensure_dir(fullfile(cfg.outRoot, 'qc'));

runCleanup = onCleanup(@() finalize_incomplete_run_v6(cfg));
runStamp = char(datetime('now','Format','yyyyMMdd_HHmmss'));
diary(fullfile(cfg.outRoot, 'logs', ['run_' runStamp '.txt']));

fprintf('\n=== ds004200 REPRODUCIBLE SCALP-ONLY PIPELINE v%s ===\n', cfg.pipelineVersion);
fprintf('Started: %s\n', char(datetime('now','Format','yyyy-MM-dd HH:mm:ss')));
fprintf('Root: %s\n', cfg.root);
fprintf('Output: %s\n\n', cfg.outRoot);

%% ===================== SUBJECT DETECTION =====================
if isempty(cfg.subjects)
    d = dir(fullfile(cfg.root, 'sub-*'));
    d = d([d.isdir]);
    cfg.subjects = {d.name};
end

fprintf('Detected %d subjects.\n', numel(cfg.subjects));
disp(cfg.subjects(:));

%% ===================== EEGLAB =====================
if exist(cfg.eeglabPath, 'dir')
    addpath(cfg.eeglabPath);
end

if exist('eeglab', 'file') ~= 2
    error('EEGLAB not found. Set cfg.eeglabPath correctly.');
end

try
    eeglab('nogui');
catch
    eeglab;
end

preflight_environment_v6(cfg);
if cfg.runSelfTests
    run_self_tests_v6(cfg);
end
inputManifest = write_run_manifest(cfg, 'STARTED');

%% ===================== BUILD BEHAVIOUR =====================
fprintf('\n=== 1) Building robust behaviour table ===\n');
behMaster = build_behavior_master(cfg);
behaviorQC = audit_behavior_master(behMaster, cfg);
writetable(behMaster, fullfile(cfg.outRoot, 'tables', 'behMaster.csv'));
writetable(behaviorQC, fullfile(cfg.outRoot, 'tables', 'behavior_subject_qc.csv'));
save(fullfile(cfg.outRoot, 'mat', 'behMaster.mat'), 'behMaster', 'behaviorQC', 'cfg');

fprintf('Behaviour rows: %d\n', height(behMaster));
fprintf('Production rows: %d\n', sum(behMaster.task == "production"));
fprintf('Perception rows: %d\n', sum(behMaster.task == "perception"));
