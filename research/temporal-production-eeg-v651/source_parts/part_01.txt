plot_behavior_qc(behMaster, cfg);

%% ===================== SUBJECT LOOP: FEATURES + TOPO + PSD =====================
fprintf('\n=== 2) Subject loop: continuous envelopes, prediction features, topography, PSD ===\n');

allTrialFeatures = table();
topoT = table();
psdT = table();
subjectQC = table();
subjectLedger = table();

representativeChanlocs = [];
representativeLabels = strings(0,1);

for s = 1:numel(cfg.subjects)
    subID = cfg.subjects{s};
    fprintf('\n--- %s (%d/%d) ---\n', subID, s, numel(cfg.subjects));

    try
        checkpointPath = fullfile(cfg.outRoot, 'mat', ['checkpoint_' char(string(subID)) '.mat']);
        checkpointContract = build_checkpoint_contract(subID, cfg, inputManifest);
        if cfg.resumeIncompleteRun && exist(checkpointPath, 'file')
            C = load(checkpointPath, 'Tfeat', 'Ttopo', 'Tpsd', 'QC', ...
                'chanlocs', 'labels', 'cfgCheckpoint', 'checkpointContract');
            if ~isfield(C,'cfgCheckpoint') || ~isequaln(C.cfgCheckpoint, cfg)
                error('v6:CheckpointConfigMismatch', ...
                    '%s checkpoint was created with a different configuration.', subID);
            end
            if ~isfield(C,'checkpointContract') || ~isequaln(C.checkpointContract, checkpointContract)
                error('v6:CheckpointProvenanceMismatch', ...
                    ['%s checkpoint does not match the current code, montage, or input hashes. ' ...
                     'Delete that checkpoint or use a new runLabel.'], subID);
            end
            Tfeat = C.Tfeat; Ttopo = C.Ttopo; Tpsd = C.Tpsd;
            QC = C.QC; chanlocs = C.chanlocs; labels = C.labels;
            fprintf('%s: loaded provenance-validated subject checkpoint.\n', subID);
        else
            [Tfeat, Ttopo, Tpsd, QC, chanlocs, labels] = run_subject_allinone(subID, behMaster, cfg);
            if cfg.checkpointSubjects
                cfgCheckpoint = cfg;
                save(checkpointPath, 'Tfeat', 'Ttopo', 'Tpsd', 'QC', ...
                    'chanlocs', 'labels', 'cfgCheckpoint', 'checkpointContract', '-v7.3');
            end
        end

        allTrialFeatures = [allTrialFeatures; Tfeat]; %#ok<AGROW>
        topoT = [topoT; Ttopo]; %#ok<AGROW>
        psdT = [psdT; Tpsd]; %#ok<AGROW>
        subjectQC = [subjectQC; QC]; %#ok<AGROW>

        if isempty(representativeChanlocs)
            representativeChanlocs = chanlocs;
            representativeLabels = labels;
        else
            validate_montage_consistency(representativeChanlocs, representativeLabels, chanlocs, labels, subID);
        end

        lr = height(subjectLedger) + 1;
        subjectLedger.subject(lr,1) = string(subID);
        subjectLedger.status(lr,1) = "included";
        subjectLedger.reason(lr,1) = "";

    catch ME
        if ~is_expected_subject_exclusion(ME)
            % Programming, provenance, dependency, configuration, and contract
            % failures are fatal. They must never masquerade as participant QC.
            diary off;
            rethrow(ME);
        end

        lr = height(subjectLedger) + 1;
        subjectLedger.subject(lr,1) = string(subID);
        subjectLedger.status(lr,1) = "excluded";
        subjectLedger.reason(lr,1) = string(ME.identifier) + ": " + string(ME.message);
        writetable(subjectLedger, fullfile(cfg.outRoot, 'tables', 'subject_inclusion_ledger.csv'));
        warning('%s excluded by prespecified subject-level QC: %s', subID, ME.message);
        for k = 1:numel(ME.stack)
            fprintf('  %s line %d\n', ME.stack(k).name, ME.stack(k).line);
        end
        if cfg.failOnSubjectError
            diary off;
            rethrow(ME);
        end
    end
end

writetable(subjectLedger, fullfile(cfg.outRoot, 'tables', 'subject_inclusion_ledger.csv'));
nIncluded = sum(subjectLedger.status == "included");
if nIncluded < cfg.minimumIncludedSubjects
    error('v6:TooFewSubjects', 'Only %d subjects passed; minimum analyzable cohort is %d.', ...
        nIncluded, cfg.minimumIncludedSubjects);
end
if nIncluded < numel(cfg.subjects)
    warning('v6:SubjectAttrition', ...
        '%d/%d subjects completed preprocessing. Inspect subject_inclusion_ledger.csv; no failure was hidden.', ...
        nIncluded, numel(cfg.subjects));
end

writetable(allTrialFeatures, fullfile(cfg.outRoot, 'tables', 'continuous_driver_trial_features.csv'));
writetable(topoT, fullfile(cfg.outRoot, 'tables', 'topomap_bandpower_by_subject_channel.csv'));
writetable(psdT, fullfile(cfg.outRoot, 'tables', 'subject_psd_spectrum.csv'));
writetable(subjectQC, fullfile(cfg.outRoot, 'tables', 'subject_qc.csv'));

save(fullfile(cfg.outRoot, 'mat', 'all_subject_outputs.mat'), ...
    'allTrialFeatures', 'topoT', 'psdT', 'subjectQC', ...
    'representativeChanlocs', 'representativeLabels', 'cfg', '-v7.3');

fprintf('\nTrial-feature rows: %d\n', height(allTrialFeatures));
fprintf('Topomap rows: %d\n', height(topoT));
fprintf('PSD rows: %d\n', height(psdT));
disp(subjectQC);

plot_feature_qc(allTrialFeatures, subjectQC, cfg);

%% ===================== TOPOGRAPHIC CONTRASTS + PSD SUMMARY =====================
fprintf('\n=== 3) Topographic contrasts + grand PSD ===\n');

contrastErrorT = compute_topo_contrast(topoT, 'highError', 'lowError', 'highError_minus_lowError');
contrastCorrectionGainT = compute_topo_contrast(topoT, 'highCorrectionGain', 'lowCorrectionGain', 'highCorrectionGain_minus_lowCorrectionGain');
grandPsdT = compute_grand_average_psd(psdT);
topoStatsT = run_primary_topography_inference(topoT, contrastErrorT, contrastCorrectionGainT, ...
    representativeChanlocs, representativeLabels, cfg);

writetable(contrastErrorT, fullfile(cfg.outRoot, 'tables', 'topomap_contrast_highError_minus_lowError.csv'));
writetable(contrastCorrectionGainT, fullfile(cfg.outRoot, 'tables', 'topomap_contrast_highCorrectionGain_minus_lowCorrectionGain.csv'));
writetable(grandPsdT, fullfile(cfg.outRoot, 'tables', 'grand_average_psd_spectrum.csv'));
writetable(topoStatsT, fullfile(cfg.outRoot, 'tables', 'primary_topography_cluster_inference.csv'));

save(fullfile(cfg.outRoot, 'mat', 'topomap_psd_contrasts.mat'), ...
    'contrastErrorT', 'contrastCorrectionGainT', 'grandPsdT', 'topoStatsT', '-v7.3');

%% ===================== PREDICTION =====================
fprintf('\n=== 4) Grouped nested LOSO prediction ===\n');

predResults = run_prediction_v65(allTrialFeatures, cfg, "original_condition");
if cfg.runStrictBlockSensitivity
    fprintf('\n=== 4b) Strict true-block sensitivity prediction ===\n');
    predStrict = run_prediction_v65(allTrialFeatures, cfg, "strict_protocol_block");
else
    predStrict = struct();
end

writetable(predResults.modelRMSE_current, fullfile(cfg.outRoot, 'tables', 'prediction_rmse_current_relError.csv'));
writetable(predResults.modelRMSE_next, fullfile(cfg.outRoot, 'tables', 'prediction_rmse_next_error.csv'));
writetable(predResults.predictionsCurrent, fullfile(cfg.outRoot, 'tables', 'prediction_fold_predictions_current.csv'));
writetable(predResults.predictionsNext, fullfile(cfg.outRoot, 'tables', 'prediction_fold_predictions_next.csv'));

writetable(predResults.contrast_current_primary, fullfile(cfg.outRoot, 'tables', 'contrasts_current_vs_history.csv'));
writetable(predResults.contrast_next_primary, fullfile(cfg.outRoot, 'tables', 'contrasts_next_vs_currentBehavior.csv'));
writetable(predResults.selectedLambdaCurrent, fullfile(cfg.outRoot, 'tables', 'selected_lambdas_current.csv'));
writetable(predResults.selectedLambdaNext, fullfile(cfg.outRoot, 'tables', 'selected_lambdas_next.csv'));

writetable(predResults.subjectRMSE_current, fullfile(cfg.outRoot, 'tables', 'subject_rmse_current_relError.csv'));
writetable(predResults.subjectRMSE_next, fullfile(cfg.outRoot, 'tables', 'subject_rmse_next_error.csv'));
writetable(predResults.tempoRMSE_current, fullfile(cfg.outRoot, 'tables', 'tempo_rmse_current_relError.csv'));
writetable(predResults.tempoRMSE_next, fullfile(cfg.outRoot, 'tables', 'tempo_rmse_next_error.csv'));

if cfg.runStrictBlockSensitivity
    writetable(predStrict.modelRMSE_current, fullfile(cfg.outRoot, 'tables', 'sensitivity_strictblock_prediction_rmse_current.csv'));
    writetable(predStrict.modelRMSE_next, fullfile(cfg.outRoot, 'tables', 'sensitivity_strictblock_prediction_rmse_next.csv'));
    writetable(predStrict.predictionsCurrent, fullfile(cfg.outRoot, 'tables', 'sensitivity_strictblock_predictions_current.csv'));
    writetable(predStrict.predictionsNext, fullfile(cfg.outRoot, 'tables', 'sensitivity_strictblock_predictions_next.csv'));
    writetable(predStrict.contrast_current_primary, fullfile(cfg.outRoot, 'tables', 'sensitivity_strictblock_contrasts_current.csv'));
    writetable(predStrict.contrast_next_primary, fullfile(cfg.outRoot, 'tables', 'sensitivity_strictblock_contrasts_next.csv'));
end

save(fullfile(cfg.outRoot, 'mat', 'predictionResults_v6_5_1_original_compatible.mat'), 'predResults', 'predStrict', '-v7.3');

fprintf('\nCurrent relError RMSE:\n');
disp(predResults.modelRMSE_current);

fprintf('\nNext-trial relative-error RMSE:\n');
disp(predResults.modelRMSE_next);

fprintf('\nCurrent relError contrasts vs history baseline:\n');
disp(predResults.contrast_current_primary);

fprintf('\nNext-error contrasts vs current-behavior baseline:\n');
disp(predResults.contrast_next_primary);

%% ===================== NEGATIVE CONTROLS =====================
if cfg.doNegativeControls
    fprintf('\n=== 5) Negative controls ===\n');
    neg = run_negative_controls(predResults.cleanedTable, cfg);

    writetable(neg.summary, fullfile(cfg.outRoot, 'tables', 'negative_controls_summary.csv'));
    writetable(neg.shuffleQC, fullfile(cfg.outRoot, 'tables', 'negative_controls_shuffle_qc.csv'));
    save(fullfile(cfg.outRoot, 'mat', 'negative_controls.mat'), 'neg', '-v7.3');

    disp(neg.summary);
else
    neg = struct();
end

%% ===================== FIGURES =====================
fprintf('\n=== 6) Creating publication-style figures ===\n');

make_psd_figures(psdT, grandPsdT, cfg);
make_topomap_figures(topoT, contrastErrorT, contrastCorrectionGainT, topoStatsT, ...
    representativeChanlocs, representativeLabels, cfg);
plot_prediction_summary(predResults, cfg);
make_subject_prediction_figures(predResults, cfg);

if cfg.doNegativeControls
    make_negative_control_figures(neg, cfg);
end

validate_outputs_v65(allTrialFeatures, topoT, psdT, subjectLedger, representativeLabels, ...
    predResults, predStrict, neg, behMaster, behaviorQC, cfg);
write_run_manifest(cfg, 'SUCCESS');

results = struct();
results.cfg = cfg;
results.behavior = behMaster;
results.behaviorQC = behaviorQC;
results.subjectLedger = subjectLedger;
results.subjectQC = subjectQC;
results.trialFeatures = allTrialFeatures;
results.topography = topoT;
results.topographyStats = topoStatsT;
results.psd = psdT;
results.prediction = predResults;
results.predictionStrictBlockSensitivity = predStrict;
results.negativeControls = neg;

fprintf('\n=== REPRODUCIBLE v6.4 ORIGINAL-COMPATIBLE PIPELINE FINISHED ===\n');
fprintf('Finished: %s\n', char(datetime('now','Format','yyyy-MM-dd HH:mm:ss')));
fprintf('Outputs saved to:\n%s\n', cfg.outRoot);
diary off;
clear runCleanup;
end

function finalize_incomplete_run_v6(cfg)
    % onCleanup handler: leave successful runs untouched; otherwise mark the
    % run FAILED so the exact runLabel is safely resumable from checkpoints.
    try
        if ~exist(cfg.outRoot,'dir'), return; end
        if exist(fullfile(cfg.outRoot,'SUCCESS'),'file'), return; end
        statusPath = fullfile(cfg.outRoot,'RUN_STATUS.txt');
        fid = fopen(statusPath,'w');
        if fid >= 0
            fprintf(fid,'FAILED\n');
            fclose(fid);
        end
        diary off;
    catch
        % Cleanup must never mask the original MATLAB exception.
    end
end

function ensure_dir(p)
    if ~exist(p, 'dir')
        mkdir(p);
    end
end

function out = merge_structs(base, override)
    out = base;
    names = fieldnames(override);
    for i = 1:numel(names)
        name = names{i};
        if isfield(base,name) && isstruct(base.(name)) && isstruct(override.(name)) && ...
                isscalar(base.(name)) && isscalar(override.(name))
            out.(name) = merge_structs(base.(name), override.(name));
        else
            out.(name) = override.(name);
        end
    end
end

function tf = is_expected_subject_exclusion(ME)
    % Only explicitly enumerated data-quality failures may be downgraded to a
    % participant exclusion. Unknown/MATLAB/programming/provenance failures are fatal.
    id = string(ME.identifier);
    allowed = [ ...
        "v4:EpochedInput", "v4:InvalidRawLabels", "v4:MissingRawChannels", ...
        "v4:UnexpectedAFzData", "v4:InvalidAFzLocation", "v4:BadMastoidStructural", ...
        "v4:NonfiniteMastoid", "v4:NonfiniteRawMastoid", ...
        "v4:TooManyRawBadChannels", "v4:TooManyBadChannels", ...
        "v4:TooFewFiniteSamplesForCausalQC", "v4:TooFewSamplesForCausalRepair", ...
        "v4:TooFewSamplesForInterpolation", "v4:TooLittleICAData", ...
        "v4:ICARank", "v4:TooManyRejectedICs", "v4:NonfiniteAfterFilter", ...
        "v62:LowAlignmentCoverage"];
    tf = any(id == allowed);
end

function validate_config_v6(cfg)
    if ~isscalar(cfg) || ~isstruct(cfg)
        error('v4:InvalidConfig', 'Configuration must be a scalar struct.');
    end
    if ~exist(cfg.root, 'dir')
        error('v4:MissingRoot', 'Dataset root does not exist: %s', cfg.root);
    end
    if cfg.targetFs <= 2*max(cfg.predBands(:))
        error('v4:Nyquist', 'targetFs must exceed twice the highest band edge.');
