    med = median(logVar, 'omitnan');
    scale = 1.4826 * median(abs(logVar - med), 'omitnan');
    if ~isfinite(scale) || scale <= 0, scale = 1; end
    z = (logVar - med) ./ scale;
    bad = missingFraction > cfg.maxMissingFractionPerChannel | ...
        channelStd < cfg.badChannelMinStdUv | abs(z) > cfg.badChannelRobustZ;
    badIdx = find(bad);
end

function [EEG, rejectedIC, icaModel] = run_iclabel_cleaning(EEG, rawGoodMask, cfg, subID)
    EEGica = pop_eegfiltnew(EEG, cfg.icaHighpassHz, []);
    rawGoodMask = logical(rawGoodMask(:));
    icaArtifactMask = make_artifact_mask(double(EEGica.data), cfg, EEGica.srate);
    icaArtifactMask = logical(icaArtifactMask(:));
    if numel(rawGoodMask) ~= EEGica.pnts || numel(icaArtifactMask) ~= EEGica.pnts
        error('v4:ICAMaskLengthMismatch', ...
            '%s ICA mask length mismatch (raw=%d, artifact=%d, pnts=%d).', ...
            subID, numel(rawGoodMask), numel(icaArtifactMask), EEGica.pnts);
    end
    goodForICA = rawGoodMask & icaArtifactMask;
    if mean(goodForICA) < 0.50
        error('v4:TooLittleICAData', '%s has fewer than 50%% clean ICA samples.', subID);
    end
    EEGica.data = EEGica.data(:,goodForICA);
    EEGica.pnts = size(EEGica.data,2);
    EEGica.trials = 1;
    EEGica.xmin = 0;
    EEGica.xmax = (EEGica.pnts-1)/EEGica.srate;
    EEGica.times = (0:EEGica.pnts-1) / EEGica.srate * 1000;
    EEGica.event = [];
    EEGica.urevent = [];
    EEGica = eeg_checkset(EEGica);

    rankSample = double(EEGica.data(:,1:min(EEGica.pnts,50000)));
    dataRank = rank(rankSample * rankSample');
    if dataRank < 2
        error('v4:ICARank', '%s ICA data rank is %d.', subID, dataRank);
    end
    previousRng = rng;
    rng(subject_specific_seed(subID, cfg.masterSeed), 'twister');
    restoreRng = onCleanup(@() rng(previousRng));
    if dataRank < EEGica.nbchan
        EEGica = pop_runica(EEGica, 'icatype','runica', 'extended',1, ...
            'pca',dataRank);
    else
        EEGica = pop_runica(EEGica, 'icatype','runica', 'extended',1);
    end
    clear restoreRng;

    EEG.icaweights = EEGica.icaweights;
    EEG.icasphere = EEGica.icasphere;
    EEG.icachansind = 1:EEG.nbchan;
    EEG.icawinv = pinv(EEG.icaweights * EEG.icasphere);
    EEG.icaact = [];
    EEG = eeg_checkset(EEG, 'ica');
    EEG = iclabel(EEG, 'default');
    if ~isfield(EEG.etc, 'ic_classification') || ...
            ~isfield(EEG.etc.ic_classification, 'ICLabel')
        error('v4:ICLabelFailure', '%s ICLabel produced no classifications.', subID);
    end
    C = EEG.etc.ic_classification.ICLabel.classifications;
    rejectedIC = find(C(:,2) >= cfg.icaMuscleThreshold | ...
        C(:,3) >= cfg.icaEyeThreshold | C(:,5) >= cfg.icaLineThreshold | ...
        C(:,6) >= cfg.icaChannelNoiseThreshold);
    if numel(rejectedIC) / size(C,1) > cfg.maxRejectedICFraction
        error('v4:TooManyRejectedICs', '%s would reject %d/%d ICs.', ...
            subID, numel(rejectedIC), size(C,1));
    end
    icaModel.unmixing = double(EEG.icaweights * EEG.icasphere);
    icaModel.mixing = pinv(icaModel.unmixing);
    if ~isempty(rejectedIC)
        EEG = pop_subcomp(EEG, rejectedIC, 0);
    end
end

function EEG = apply_ica_rejection_model(EEG, icaModel, rejectedIC, subID)
    if isempty(rejectedIC)
        return;
    end
    X = double(EEG.data);
    if size(icaModel.unmixing,2) ~= size(X,1) || ...
            size(icaModel.mixing,1) ~= size(X,1)
        error('v4:CausalICADimension', '%s ICA model does not match the causal branch.', subID);
    end
    activations = icaModel.unmixing * X;
    activations(rejectedIC,:) = 0;
    Xclean = icaModel.mixing * activations;
    if any(~isfinite(Xclean(:)))
        error('v4:CausalICAReconstruction', '%s causal ICA reconstruction is nonfinite.', subID);
    end
    EEG.data = cast(Xclean, 'like', EEG.data);
    EEG = eeg_checkset(EEG);
end

function seed = subject_specific_seed(subID, baseSeed)
    codes = double(char(string(subID)));
    weights = 1:numel(codes);
    seed = mod(double(baseSeed) + sum(codes .* weights), 2^32-1);
    if seed < 1, seed = 1; end
end

function validate_scalp_layout(chanlocs, expectedLabels, subID)
    labels = lower(strtrim(string({chanlocs.labels})))';
    if ~isequal(labels, lower(expectedLabels(:)))
        error('v4:ScalpOrder', '%s scalp labels are not in canonical order.', subID);
    end
    xy = chanlocs_xy(chanlocs);
    if any(~isfinite(xy(:)))
        error('v4:MissingCoordinates', '%s has missing scalp coordinates.', subID);
    end
    for i = 1:size(xy,1)
        d = sqrt(sum((xy - xy(i,:)).^2,2));
        d(i) = inf;
        if min(d) < 1e-6
            error('v4:DuplicateCoordinates', '%s has duplicate scalp coordinates near %s.', ...
                subID, expectedLabels(i));
        end
    end
end

function xy = chanlocs_xy(chanlocs)
    n = numel(chanlocs);
    xy = nan(n,2);
    for i = 1:n
        if isfield(chanlocs(i),'theta') && isfield(chanlocs(i),'radius') && ...
                ~isempty(chanlocs(i).theta) && ~isempty(chanlocs(i).radius)
            th = deg2rad(double(chanlocs(i).theta));
            r = double(chanlocs(i).radius);
            xy(i,:) = [r*sin(th), r*cos(th)];
        elseif isfield(chanlocs(i),'X') && isfield(chanlocs(i),'Y') && ...
                ~isempty(chanlocs(i).X) && ~isempty(chanlocs(i).Y)
            xy(i,:) = [double(chanlocs(i).X), double(chanlocs(i).Y)];
        end
    end
end

function validate_montage_consistency(refLocs, refLabels, locs, labels, subID)
    if ~isequal(lower(refLabels(:)), lower(labels(:)))
        error('v4:CrossSubjectLabels', '%s montage labels differ from the canonical subject.', subID);
    end
    a = chanlocs_xy(refLocs);
    b = chanlocs_xy(locs);
    if ~isequal(size(a),size(b)) || max(abs(a(:)-b(:)),[],'omitnan') > 1e-4
        error('v4:CrossSubjectCoordinates', '%s montage coordinates differ from the canonical subject.', subID);
    end
end

function behMaster = build_behavior_master(cfg)
    behMaster = table();

    for s = 1:numel(cfg.subjects)
        subID = cfg.subjects{s};
        behFile = fullfile(cfg.root, subID, 'beh', [subID '_task-temporalscaling_beh.tsv']);

        if ~exist(behFile, 'file')
            warning('v64:MissingBehaviourFile', 'Missing behaviour file: %s', behFile);
            continue;
        end

        Raw = readtable(behFile, 'FileType','text', 'Delimiter','\t');

        required = {'block','trial','fbCondition','responseTime'};
        for r = 1:numel(required)
            if ~ismember(required{r}, Raw.Properties.VariableNames)
                error('v64:MissingBehaviourColumn', '%s missing required behaviour column: %s', subID, required{r});
            end
        end

        n = height(Raw);
        T = table();
        T.subject = repmat(string(subID), n, 1);
        T.rawRow = (1:n)';

        % OFFICIAL BIDS SEMANTICS FROM THE ORIGINAL DATASET:
        % block = block TYPE, not a unique physical block identifier.
        % 1 production-fast, 2 production-medium, 3 production-slow,
        % 4 perception-fast, 5 perception-medium, 6 perception-slow.
        T.block = to_num(Raw.block);
        T.conditionCode = T.block; % explicit semantic alias for modelling/QC
        T.trial = to_num(Raw.trial); % official within-block trial number
        T.fbCondition = to_num(Raw.fbCondition);
        T.responseTime = to_num(Raw.responseTime);

        if any(~isfinite(T.block)) || any(T.block ~= round(T.block)) || ...
                any(~ismember(T.block, 1:6))
            error('v64:BlockTypeContract', ...
                '%s BIDS block values must be finite integers in 1..6.', subID);
        end
        if any(~isfinite(T.trial)) || any(T.trial ~= round(T.trial)) || ...
                any(T.trial < 1) || any(T.trial > cfg.trialsPerProtocolBlock)
            error('v64:TrialNumberContract', ...
                '%s official trial numbers must be integers in 1..%d.', subID, cfg.trialsPerProtocolBlock);
        end

        T.task = strings(n,1);
        T.task(T.block <= 3) = "production";
        T.task(T.block >= 4) = "perception";

        T.tempo = nan(n,1);
        T.tempo(T.block == 1 | T.block == 4) = 1;
        T.tempo(T.block == 2 | T.block == 5) = 2;
        T.tempo(T.block == 3 | T.block == 6) = 3;

        T.blockLabel = strings(n,1);
        T.blockLabel(T.block == 1) = "production_fast";
        T.blockLabel(T.block == 2) = "production_medium";
        T.blockLabel(T.block == 3) = "production_slow";
        T.blockLabel(T.block == 4) = "perception_fast";
        T.blockLabel(T.block == 5) = "perception_medium";
        T.blockLabel(T.block == 6) = "perception_slow";

        T.tempoLabel = strings(n,1);
        T.tempoLabel(T.tempo == 1) = "short_fast";
        T.tempoLabel(T.tempo == 2) = "medium";
        T.tempoLabel(T.tempo == 3) = "long_slow";

        T.targetDuration = nan(n,1);
        for tempo = 1:3
            T.targetDuration(T.tempo == tempo) = cfg.targetDurationsSec(tempo);
        end

        idxProd = T.task == "production";
        idxPerc = T.task == "perception";
        T.prodSeq = nan(n,1);
        T.perceptionSeq = nan(n,1);
        pidx = find(idxProd);
        qidx = find(idxPerc);
        T.prodSeq(pidx) = (1:numel(pidx))';
        T.perceptionSeq(qidx) = (1:numel(qidx))';

        % Original-paper sequence index: after selecting one BIDS block TYPE,
        % rows remain in their original acquisition order and are concatenated
        % across repeated physical blocks. This reproduces analysis_02_beh.m.
        T.conditionTrialSeq = nan(n,1);
        for code = 1:6
            cidx = find(T.block == code);
            T.conditionTrialSeq(cidx) = (1:numel(cidx))';
        end

        % Strict sensitivity: reconstruct true 10-trial physical blocks from the
        % OFFICIAL trial-number column. Unlike v6.4, this does not infer boundaries
        % merely by counting rows; a reset to trial==1 defines a new block and
        % every block must be an intact 1..10 sequence with one block type.
        T.protocolBlock = nan(n,1);
        T.withinBlockSeq = T.trial;
        T.protocolBlock(idxProd) = reconstruct_task_protocol_blocks( ...
            T.trial(idxProd), T.block(idxProd), subID, "production", cfg);
        T.protocolBlock(idxPerc) = reconstruct_task_protocol_blocks( ...
            T.trial(idxPerc), T.block(idxPerc), subID, "perception", cfg);
        validate_protocol_blocks(T, subID, cfg);

        T.producedDuration = nan(n,1);
        T.producedDuration(idxProd) = T.responseTime(idxProd);
        T.signedError = nan(n,1);
        T.absError = nan(n,1);
        T.relError = nan(n,1);
        T.absRelError = nan(n,1);
        T.signedError(idxProd) = T.producedDuration(idxProd) - T.targetDuration(idxProd);
        T.absError(idxProd) = abs(T.signedError(idxProd));
        T.relError(idxProd) = T.signedError(idxProd) ./ T.targetDuration(idxProd);
        T.absRelError(idxProd) = abs(T.relError(idxProd));
        T.behaviorDurationValid = idxProd & ...
            valid_production_duration(T.producedDuration, T.tempo, cfg);

        % OFFICIAL production feedback labels from dataset metadata:
        % 3=early, 4=on-time, 5=late. Codes 1/2 belong to perception and are
        % NEVER reconstructed from the timing outcome. If non-official codes
        % appear in production rows, feedback-dependent analyses are disabled
        % for that subject while EEG-only endpoints remain eligible.
        T.feedbackLabel = strings(n,1);
        T.feedbackSigned = nan(n,1);
        T.feedbackCodeScheme = repmat("not_applicable", n, 1);
        prodCodes = unique(T.fbCondition(idxProd & isfinite(T.fbCondition)));
        if isempty(prodCodes)
            T.feedbackCodeScheme(idxProd) = "missing";
            warning('v65:MissingFeedbackCodes', '%s has no finite production feedback codes.', subID);
        elseif all(ismember(prodCodes, [3 4 5]))
            T.feedbackLabel(idxProd & T.fbCondition == 3) = "early";
            T.feedbackSigned(idxProd & T.fbCondition == 3) = -1;
            T.feedbackLabel(idxProd & T.fbCondition == 4) = "on_time";
            T.feedbackSigned(idxProd & T.fbCondition == 4) = 0;
            T.feedbackLabel(idxProd & T.fbCondition == 5) = "late";
            T.feedbackSigned(idxProd & T.fbCondition == 5) = 1;
            T.feedbackCodeScheme(idxProd) = "official_345";
        else
            T.feedbackCodeScheme(idxProd) = "nonofficial_unusable";
            if cfg.requireOfficialProductionFeedback
                warning('v65:NonOfficialProductionFeedback', ...
                    ['%s production fbCondition contains non-official code(s) %s. ' ...
                     'Official metadata defines production feedback only as 3/4/5; ' ...
                     'feedback-dependent covariates are disabled rather than reconstructed.'], ...
                    subID, mat2str(prodCodes(:)'));
            end
        end

        behMaster = [behMaster; T]; %#ok<AGROW>
    end

    if isempty(behMaster)
        error('v64:EmptyBehaviourMaster', 'No behavioural rows were loaded.');
