    tf = true;
    if isempty(T), return; end
    blocks = unique(T.protocolBlock(isfinite(T.protocolBlock)));
    for bi = 1:numel(blocks)
        B = T(T.protocolBlock == blocks(bi), :);
        if height(B) ~= cfg.trialsPerProtocolBlock || ...
                numel(unique(B.withinBlockSeq)) ~= cfg.trialsPerProtocolBlock || ...
                ~isequal(sort(B.withinBlockSeq), (1:cfg.trialsPerProtocolBlock)') || ...
                numel(unique(B.tempo)) ~= 1 || numel(unique(B.conditionCode)) ~= 1
            tf = false;
            return;
        end
    end
end

function tf = valid_production_duration(durationSec, tempo, cfg)
    % Original-paper production QC (analysis_02_beh.m):
    % doLim = [0.2, 0.8*5; 0.2, 1.65*5; 0.2, 2.5*5].
    durationSec = double(durationSec);
    tempo = double(tempo);
    if ~isequal(size(durationSec), size(tempo))
        if isscalar(durationSec)
            durationSec = repmat(durationSec, size(tempo));
        elseif isscalar(tempo)
            tempo = repmat(tempo, size(durationSec));
        else
            error('v65:DurationShapeContract', 'durationSec and tempo must have compatible sizes.');
        end
    end
    tf = false(size(durationSec));
    for t = 1:3
        use = tempo == t & isfinite(durationSec);
        tf(use) = durationSec(use) >= cfg.minTrialDurSec & ...
            durationSec(use) <= cfg.maxTrialDurSecByTempo(t);
    end
end

function x = to_num(x)
    if isnumeric(x)
        x = double(x);
        return;
    end

    if iscell(x), x = string(x); end
    if iscategorical(x), x = string(x); end

    x = string(x);
    x = strrep(x, ',', '.');
    x = standardizeMissing(x, ["", "NA", "NaN", "nan", "n/a", "N/A", "None", "missing"]);
    x = str2double(x);
end

function plot_behavior_qc(behMaster, cfg)
    figDir = fullfile(cfg.outRoot, 'figures');

    idx = behMaster.task == "production" & behMaster.behaviorDurationValid & isfinite(behMaster.relError);
    if ~any(idx), return; end

    f = figure('Color','w','Name','Figure 1 behaviour relative error');
    boxchart(categorical(behMaster.tempoLabel(idx)), behMaster.relError(idx));
    yline(0,'--');
    xlabel('Tempo');
    ylabel('Relative error');
    title('Fig1: Behavioural timing error by target interval');
    grid on;
    saveas(f, fullfile(figDir, 'Fig1_behavior_relError_by_tempo.png'));
    close(f);

    % Exact original-paper adjustment variable: within each production
    % condition, successor RT minus that condition's mean RT, grouped by the
    % feedback on the current observation (analysis_02_beh.m semantics).
    valid = behMaster.task == "production" & ...
        isfinite(behMaster.paperNextAdjustmentSec) & strlength(behMaster.feedbackLabel) > 0;
    if any(valid)
        f = figure('Color','w','Name','Figure 1 original-paper adjustment by feedback');
        boxchart(categorical(behMaster.feedbackLabel(valid)), behMaster.paperNextAdjustmentSec(valid));
        yline(0,'--');
        xlabel('Production feedback');
        ylabel('Successor RT - condition mean RT (s)');
        title('Fig1: Original-paper feedback-linked successor adjustment');
        grid on;
        saveas(f, fullfile(figDir, 'Fig1_behavior_paperAdjustment_by_feedback.png'));
        close(f);
    end
end

function [Tfeat, Ttopo, Tpsd, QC, chanlocs, labels] = run_subject_allinone(subID, behMaster, cfg)
    Tfeat = table();
    QC = table();

    setFile = fullfile(cfg.root, subID, 'eeg', [subID '_task-temporalscaling_eeg.set']);
    if ~exist(setFile, 'file')
        error('Missing EEG file: %s', setFile);
    end

    [fp, fn, ext] = fileparts(setFile);
    EEG = pop_loadset('filename', [fn ext], 'filepath', fp);
    EEG = eeg_checkset(EEG);

    originalFs = EEG.srate;

    fprintf('Loaded %s | channels=%d | fs=%.1f | duration=%.1fs | events=%d\n', ...
        subID, EEG.nbchan, EEG.srate, EEG.pnts/EEG.srate, numel(EEG.event));

    [EEG, causalData, causalGoodMask, causalFs, prepQC, rawGoodMask] = ...
        preprocess_subject_eeg(EEG, subID, cfg);
    fs = EEG.srate;

    labels = get_chan_labels(EEG);
    chanlocs = EEG.chanlocs;

    assert(EEG.nbchan == numel(cfg.scalpLabels), ...
        'v4:ScalpCount', '%s: expected %d scalp channels, found %d.', ...
        subID, numel(cfg.scalpLabels), EEG.nbchan);
    assert(~any(ismember(lower(labels), lower(string(cfg.auxLabels)))), ...
        'v4:AuxiliaryLeakage', '%s: auxiliary channel survived preprocessing.', subID);

    roiIdx = find_channels(labels, cfg.roiLabels);
    assert(numel(roiIdx) == numel(cfg.roiLabels), ...
        'v4:MissingROI', '%s: incomplete ROI montage; refusing all-channel fallback.', subID);

    rawGoodMask = logical(rawGoodMask(:));
    finalArtifactMask = make_artifact_mask(EEG.data, cfg, fs);
    finalArtifactMask = logical(finalArtifactMask(:));
    if numel(rawGoodMask) ~= EEG.pnts || numel(finalArtifactMask) ~= EEG.pnts
        error('v4:FinalMaskLengthMismatch', ...
            '%s final mask length mismatch (raw=%d, artifact=%d, pnts=%d).', ...
            subID, numel(rawGoodMask), numel(finalArtifactMask), EEG.pnts);
    end
    goodMask = rawGoodMask & finalArtifactMask;
    goodSamplePct = 100 * mean(goodMask);
    spectralData = interpolate_masked_samples(double(EEG.data), goodMask, subID);

    ev = parse_events(EEG);
    prodTrials = build_production_trials_from_events(ev, cfg, fs);

    B = behMaster(behMaster.subject == string(subID) & behMaster.task == "production", :);
    B = sortrows(B, 'prodSeq');

    [Ealigned, Baligned, alignQC] = align_eeg_behavior_trials(prodTrials, B, cfg);

    behaviorAlignmentCoverage = height(Ealigned) / max(height(B),1);
    eegAlignmentCoverage = height(Ealigned) / max(height(prodTrials),1);
    minCoverage = min(behaviorAlignmentCoverage, eegAlignmentCoverage);
    fprintf(['%s: production EEG=%d | aligned=%d | skipped EEG=%d | skipped beh=%d | ' ...
        'feedback known=%d agree=%d disagree=%d (%.2f%%) | coverage beh=%.1f%% EEG=%.1f%% | good samples=%.1f%%\n'], ...
        subID, height(prodTrials), height(Ealigned), alignQC.nSkippedEEG, alignQC.nSkippedBehavior, ...
        alignQC.nFeedbackKnown, alignQC.nFeedbackAgree, alignQC.nFeedbackDisagree, ...
        100*alignQC.feedbackDisagreementRate, 100*behaviorAlignmentCoverage, ...
        100*eegAlignmentCoverage, goodSamplePct);

    if height(Ealigned) == 0 || minCoverage < cfg.minAlignmentCoverage
        error('v62:LowAlignmentCoverage', ...
            '%s alignment coverage failed the hard gate (behavior %.1f%%, EEG %.1f%%; minimum %.1f%%).', ...
            subID, 100*behaviorAlignmentCoverage, 100*eegAlignmentCoverage, 100*cfg.minAlignmentCoverage);
    elseif minCoverage < cfg.minAlignmentCoverageWarn
        warning('v62:AlignmentCoverageWarning', ...
            '%s alignment coverage is below the warning threshold (behavior %.1f%%, EEG %.1f%%; warning %.1f%%).', ...
            subID, 100*behaviorAlignmentCoverage, 100*eegAlignmentCoverage, 100*cfg.minAlignmentCoverageWarn);
    end

    % Behavioural fbCondition (official 3/4/5 production labels) is the
    % authoritative source for feedback covariates. EEG feedback triggers are
    % used only as an independent validation channel when they are available.
    % Absence/sparsity of EEG feedback events must NOT erase valid behavioural
    % labels, otherwise every subject can become ineligible for prediction.
    nBehaviorFeedbackKnown = sum(isfinite(Baligned.feedbackSigned));
    feedbackModelUsable = nBehaviorFeedbackKnown >= cfg.minTrialsPerSubject;
    feedbackValidationStatus = "not_available";

    if ~feedbackModelUsable
        warning('v651:InsufficientBehaviorFeedback', ...
            ['%s has only %d finite official behavioural production feedback labels. ' ...
             'Feedback-dependent prediction covariates are disabled for this subject.'], ...
            subID, nBehaviorFeedbackKnown);
        feedbackValidationStatus = "behavior_unusable";
    elseif alignQC.nFeedbackKnown >= cfg.minFeedbackValidationTrials
        feedbackValidationStatus = "validated";
        if alignQC.feedbackDisagreementRate > cfg.maxFeedbackDisagreementRate
            feedbackModelUsable = false;
            feedbackValidationStatus = "failed";
            warning('v651:FeedbackUnreliableForPrediction', ...
                ['%s independent post-alignment feedback QC failed: %d/%d comparable labels disagree ' ...
                 '(%.2f%% > %.2f%%). Official behavioural feedback covariates are disabled for prediction, ' ...
                 'while feedback-independent EEG/topography/PSD endpoints are retained.'], ...
                subID, alignQC.nFeedbackDisagree, alignQC.nFeedbackKnown, ...
                100*alignQC.feedbackDisagreementRate, 100*cfg.maxFeedbackDisagreementRate);
        end
    elseif alignQC.nFeedbackKnown > 0
        feedbackValidationStatus = "sparse";
        warning('v651:SparseFeedbackValidation', ...
            ['%s has only %d independently comparable EEG/behavior feedback labels. ' ...
             'Official behavioural 3/4/5 labels remain usable; EEG-trigger validation is reported as sparse.'], ...
            subID, alignQC.nFeedbackKnown);
    else
        feedbackValidationStatus = "not_available";
        warning('v651:NoFeedbackValidation', ...
            ['%s has no independently comparable EEG/behavior feedback labels. ' ...
             'Official behavioural 3/4/5 labels remain usable; EEG-trigger validation is unavailable.'], subID);
    end

    Trials = table();
    Trials.subject = repmat(string(subID), height(Ealigned), 1);
    Trials.eegProdSeq = Ealigned.eegProdSeq;
    Trials.behProdSeq = Baligned.prodSeq;
    Trials.block = Baligned.block; % OFFICIAL BIDS block type (1..3 for production)
    Trials.blockLabel = Baligned.blockLabel;
    Trials.conditionCode = Baligned.conditionCode;
    Trials.trial = Baligned.trial;
    Trials.conditionTrialSeq = Baligned.conditionTrialSeq;
    Trials.protocolBlock = Baligned.protocolBlock; % strict sensitivity only
    Trials.rawRow = Baligned.rawRow;
    Trials.withinBlockSeq = Baligned.withinBlockSeq;
    Trials.tempo = Baligned.tempo;
    Trials.tempoLabel = Baligned.tempoLabel;
    Trials.startSample = Ealigned.startSample;
    Trials.respSample = Ealigned.respSample;
    Trials.targetDuration = Baligned.targetDuration;
    Trials.producedDuration = Baligned.producedDuration;
    Trials.producedDurationEEG = Ealigned.producedDurationEEG;
    Trials.durationMismatchSec = abs(Trials.producedDuration - Trials.producedDurationEEG);
    Trials.isDurationMatched = double(Trials.durationMismatchSec <= cfg.maxDurationMismatchSec);
    Trials.feedbackSignedRaw = Baligned.feedbackSigned;
    Trials.feedbackSigned = Baligned.feedbackSigned;
    Trials.feedbackLabel = Baligned.feedbackLabel;
    if ismember('feedbackSignedEEG', Ealigned.Properties.VariableNames)
        Trials.feedbackSignedEEG = Ealigned.feedbackSignedEEG;
    else
        Trials.feedbackSignedEEG = nan(height(Trials),1);
    end
    Trials.feedbackComparable = isfinite(Trials.feedbackSignedEEG) & isfinite(Trials.feedbackSigned);
    Trials.feedbackAgreement = false(height(Trials),1);
    Trials.feedbackAgreement(Trials.feedbackComparable) = ...
        Trials.feedbackSignedEEG(Trials.feedbackComparable) == Trials.feedbackSigned(Trials.feedbackComparable);
    Trials.relError = Baligned.relError;
    Trials.absRelError = Baligned.absRelError;
    Trials.currentBehaviorValid = logical(Baligned.behaviorDurationValid);
    % Primary original-paper condition-sequence adjacency.
    Trials.prevProducedDuration = Baligned.prevProducedDuration;
    Trials.prevRelError = Baligned.prevRelError;
    Trials.prevFeedbackSignedRaw = Baligned.prevFeedbackSigned;
    Trials.prevFeedbackSigned = Baligned.prevFeedbackSigned;
    Trials.prevBehaviorValid = logical(Baligned.prevBehaviorValid);
    Trials.nextProducedDuration = Baligned.nextProducedDuration;
    Trials.nextRelError = Baligned.nextRelError;
    Trials.nextRelDelta = Baligned.nextRelDelta;
    Trials.correctionGain = Baligned.correctionGain;
    Trials.nextBehaviorValid = logical(Baligned.nextBehaviorValid);
    Trials.paperNextAdjustmentSec = Baligned.paperNextAdjustmentSec;
    % Strict true-10-trial-block sensitivity adjacency.
    Trials.strictPrevProducedDuration = Baligned.strictPrevProducedDuration;
    Trials.strictPrevRelError = Baligned.strictPrevRelError;
    Trials.strictPrevFeedbackSignedRaw = Baligned.strictPrevFeedbackSigned;
    Trials.strictPrevFeedbackSigned = Baligned.strictPrevFeedbackSigned;
    Trials.strictPrevBehaviorValid = logical(Baligned.strictPrevBehaviorValid);
    Trials.strictNextProducedDuration = Baligned.strictNextProducedDuration;
    Trials.strictNextRelError = Baligned.strictNextRelError;
    Trials.strictNextRelDelta = Baligned.strictNextRelDelta;
    Trials.strictCorrectionGain = Baligned.strictCorrectionGain;
    Trials.strictNextBehaviorValid = logical(Baligned.strictNextBehaviorValid);
    Trials.feedbackCodeScheme = Baligned.feedbackCodeScheme;
    Trials.feedbackModelUsable = repmat(logical(feedbackModelUsable), height(Trials), 1);
    Trials.feedbackValidationStatus = repmat(string(feedbackValidationStatus), height(Trials), 1);
    if ~feedbackModelUsable
        % Preserve raw behavioural labels for audit/provenance, but prevent
        % their use as model covariates when behavioural labels are insufficient
        % or independent validation explicitly fails.
        Trials.feedbackSigned(:) = NaN;
        Trials.prevFeedbackSigned(:) = NaN;
        Trials.strictPrevFeedbackSigned(:) = NaN;
    end

    nAlignedBeforeMismatch = height(Trials);
    mismatchRate = mean(Trials.isDurationMatched ~= 1, 'omitnan');
    % Alignment itself enforces the same duration tolerance. Any post-alignment
    % mismatch therefore indicates an implementation/rounding contract failure.
    if any(Trials.isDurationMatched ~= 1)
        error('v62:PostAlignmentDurationContract', ...
            '%s contains %d aligned pairs exceeding the prespecified %.3f-s tolerance.', ...
            subID, sum(Trials.isDurationMatched ~= 1), cfg.maxDurationMismatchSec);
    end

    Trials.baselineGoodFraction = nan(height(Trials),1);
    Trials.intervalGoodFraction = nan(height(Trials),1);
    Trials.fixedWindowValid = false(height(Trials),1);
    Trials.eegQualityPass = false(height(Trials),1);
    for tr = 1:height(Trials)
        st = Trials.startSample(tr);
        rp = Trials.respSample(tr);
        baseRows = baseline_rows(st, cfg, numel(goodMask), fs);
        intRows = st:rp;
        intRows = intRows(intRows >= 1 & intRows <= numel(goodMask));
        if isempty(baseRows) || isempty(intRows), continue; end
        Trials.baselineGoodFraction(tr) = mean(goodMask(baseRows));
        Trials.intervalGoodFraction(tr) = mean(goodMask(intRows));
        fixedEnd = st + round(cfg.fixedEarlyWinSec(2) * fs);
        % Descriptive 0-500 ms endpoint only. Primary current-error modelling
        % uses the pre-start EEG window below and is never selected by rp.
        Trials.fixedWindowValid(tr) = rp > fixedEnd;
        Trials.eegQualityPass(tr) = ...
